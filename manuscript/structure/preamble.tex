% --- Unicode math-symbol safety net ---
% pdflatex+inputenc errors FATALLY ("Unicode character ... not set up for use
% with LaTeX") on a raw Unicode math glyph that slips into source prose (e.g. a
% copy-pasted "phasor↔real-space"). Source should use the LaTeX form
% ($\leftrightarrow$, $\to$, ...); this map is the belt-and-suspenders safety
% net so a stray copy compiles instead of halting the build — load-bearing
% because this preamble + the backmatter are symlinked into every satellite
% repo (AVE-Fusion, etc.), so one stray glyph breaks them all at once.
% Scope is deliberately limited to arrows + binary relations (the recurring
% paste-risk class). Greek letters and box-drawing glyphs are NOT mapped: they
% appear legitimately in verbatim output and %-comment banners, and a stray one
% in text mode should fail loud rather than silently render. The LaTeX kernel
% already sets up → and ← (so those never erred — only ↔ did); we map just the
% glyphs the kernel leaves undefined, which also keeps the build log free of
% newunicodechar "Redefining" warnings.
\usepackage{newunicodechar}
\newunicodechar{↔}{\ensuremath{\leftrightarrow}}
\newunicodechar{⇒}{\ensuremath{\Rightarrow}}
\newunicodechar{⇐}{\ensuremath{\Leftarrow}}
\newunicodechar{⇔}{\ensuremath{\Leftrightarrow}}
\newunicodechar{↦}{\ensuremath{\mapsto}}
\newunicodechar{≤}{\ensuremath{\leq}}
\newunicodechar{≥}{\ensuremath{\geq}}
\newunicodechar{≠}{\ensuremath{\neq}}
\newunicodechar{≈}{\ensuremath{\approx}}
\newunicodechar{≡}{\ensuremath{\equiv}}
\newunicodechar{∝}{\ensuremath{\propto}}
\newunicodechar{∞}{\ensuremath{\infty}}

% --- Line-breaking discipline ---
% Global \sloppy was removed 2026-05-28 (vol9-corpus LaTeX-formatting pass):
% the long inline \texttt{} path citations that produced cosmetic Overfull
% \hbox overruns are now wrapped in \kbleaf{} (= \texttt{\seqsplit{}}, see
% commands.tex), which breaks the path at any character. With those sites
% handled at the source, the global glue-stretching of \sloppy is no longer
% needed and normal (tighter) interword spacing is restored corpus-wide.
% \emergencystretch is retained as a bounded safety net: pdfTeX may stretch
% by this amount per line before resorting to an overrun box. The reduced
% value keeps the post-build margin gate (src/scripts/check_latex_margins.py)
% tight while tolerating the few remaining narrative run-ins.
\setlength{\emergencystretch}{2em}
